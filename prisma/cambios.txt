model WeekSchedule {
  id                   Int      @id @default(autoincrement())
  idTeacherAssignment  Int     //relación con la asignatura que imparte un profesor en un grupo durante un año académico
  teacherAssignment    TeacherOnSubjectOnGroup @relation(fields: [idTeacherAssignment], references: [id],name:"TeacherAssignment", onDelete: Cascade)
  weekDay            Int      // 1=Lunes, 7=Domingo o enum de los dias??
  startTime          String   // "09:15" o usar DateTime horaInicio
  finishTime             String   // "10:15" horaFin
  //aula                 String?
  
  
  createdAt            DateTime @default(now())
  
  sessions             SessionClass[]   //las sesiones de clase de por ejemplo los lunes a primera, o los miercoles a tercera
  
  @@index([idTeacherAssignment, weekDay])//filtra por id de teacherOnSubjectOnGroup y el dia de la semana
}
//una clase propiamente hablando
model SessionClass {
  id          Int      @id @default(autoincrement())
  idSchedule   Int          //cada sesion de clase tiene un id del horario al que pertenece
  schedule     WeekSchedule @relation(fields: [idSchedule], references: [id], onDelete: Cascade)//horario
  
  date       DateTime //esto vuelve a date time
  status      String   @default("PROGRAMADA") // PROGRAMADA, REALIZADA, CANCELADA estado
  apointments String?//detalles, observaciones
  
  createdAt   DateTime @default(now())
  
  assistances Assistance[]    //las asistencias de cada alumno en esa sesión
  
  @@index([date])
  @@index([idSchedule, date])
}

model Assistance {
  id                  Int      @id @default(autoincrement())
  idSession            Int    
  session              SessionClass @relation(fields: [idSession], references: [id], onDelete: Cascade)
  
  idStudentEnrollment Int
 studentEnrollment   StudentOnSubjectOnGroup @relation(fields: [idStudentEnrollment], references: [id], name: "StudentEnrollment", onDelete: Cascade)

  
  status              AssistanceStatus @default(PRESENT)   // PRESENTE, AUSENTE, RETRASO, JUSTIFICADA
  justificationUri    String?             // URL/path del archivo subido
  justificationStatus JustificationStatus? // null = sin justificante enviado o estaba presente
  
  createdAt           DateTime @default(now())
  
  @@unique([idSession, idStudentEnrollment]) // Un alumno solo puede tener una asistencia por sesión
  @@index([idStudentEnrollment])//filtra por id StudentOnSubjectOnGroup
  @@index([idSession, status])//filtra por estado y clase
}

enum TaskType {
  PRACTICE     // Práctica
  THEORY       // Teoría/trabajo/material
  EXAM         // Examen
  PROJECT      // Proyecto
  HOMEWORK     // Deberes
}

model Task {
  id                   Int      @id @default(autoincrement())
  
  idTeacherAssignment  Int
  teacherAssignment    TeacherOnSubjectOnGroup @relation(fields: [idTeacherAssignment], references: [id], onDelete: Cascade)
  
  title                String
  description          String?
  type                 TaskType
  
  startDate            DateTime  // Fecha/hora desde que se puede ver
  dueDate              DateTime  // Fecha/hora límite de entrega

  attachmentUrl        String?//url del fichero que sube el profesor, puede ser temario o contenido para una tarea
  
  schoolYear           String    // "2024-2025"
  
  createdAt            DateTime @default(now())
  
  studentTasks         StudentTask[]  // Entregas de los alumnos
  
  @@index([idTeacherAssignment, schoolYear])
  @@index([dueDate])
}

enum SubmissionStatus {
  PENDING      // Pendiente de entregar
  SUBMITTED    // Entregada
  LATE         // Entregada tarde
  GRADED       // Calificada
  NOT_SUBMITTED // No entregada (después de la fecha)
}

model StudentTask {
  id                  Int      @id @default(autoincrement())
  
  idTask              Int
  task                Task @relation(fields: [idTask], references: [id], onDelete: Cascade)
  
  idStudentEnrollment Int
  studentEnrollment   StudentOnSubjectOnGroup @relation(fields: [idStudentEnrollment], references: [id], onDelete: Cascade)
  
  status              SubmissionStatus @default(PENDING)
  submissionDate      DateTime?        // Fecha/hora de entrega (null si no ha entregado)
  score               Decimal?         // Nota (null si no está calificada)
  feedback            String?          // Comentarios del profesor
  attachmentUrl       String?          // URL del archivo subido (opcional)
  
  createdAt           DateTime @default(now())
  
  @@unique([idTask, idStudentEnrollment])  // Un alumno solo puede tener una entrega por tarea
  @@index([idStudentEnrollment, status])
  @@index([idTask, status])
}
